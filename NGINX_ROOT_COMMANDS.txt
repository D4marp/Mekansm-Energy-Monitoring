# Setup Nginx dengan SSL - Perintah Root User (Tanpa Sudo)
# Port 80 (HTTP) + Port 443 (HTTPS)

## STEP 1: Install nginx & certbot
apt update
apt install -y nginx certbot python3-certbot-nginx

## STEP 2: Copy konfigurasi nginx
## (Dari local machine, jalankan SCP)
## scp nginx/mekansm-energy.conf root@103.27.207.58:/tmp/

## STEP 3: Copy config ke nginx directory
cp /tmp/mekansm-energy.conf /etc/nginx/sites-available/

## STEP 4: Enable site
ln -s /etc/nginx/sites-available/mekansm-energy.conf /etc/nginx/sites-enabled/
rm -f /etc/nginx/sites-enabled/default

## STEP 5: Test & start nginx (port 80 dulu)
nginx -t
systemctl restart nginx
systemctl enable nginx

## STEP 6: Setup SSL Certificate dengan Let's Encrypt
## Pilih OPTION A atau OPTION B dibawah:

# OPTION A: Untuk IP Address (Wildcard tidak bisa)
certbot certonly --standalone -d 103.27.207.58 --non-interactive --agree-tos -m admin@example.com

# OPTION B: Untuk Domain Name (recommended)
certbot certonly --nginx -d yourdomain.com --non-interactive --agree-tos -m admin@example.com

## STEP 7: Update nginx config dengan domain/IP yang benar
# Edit /etc/nginx/sites-available/mekansm-energy.conf dan sesuaikan:
# - server_name 103.27.207.58;  (ganti dengan domain/IP Anda)
# - ssl_certificate path
# - ssl_certificate_key path

nano /etc/nginx/sites-available/mekansm-energy.conf

## STEP 8: Test & reload nginx
nginx -t
systemctl reload nginx

## STEP 9: Verify status
systemctl status nginx
netstat -tlnp | grep nginx
